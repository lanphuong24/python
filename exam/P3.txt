Câu 17. Đoạn chương trình sau in ra giá trị nào?
def palindrome(s):
    left = 0
    right = len(s) - 1
    while left < right:
        if s[left] != s[right]:
            return False
        left += 1
        right -= 1
    return True

print(palindrome("level"))
A. True    B. False    C. level    D. 5

    le = []
    chan = []
    for x in a:
        if x % 2 != 0:

        else:

Câu 24. Đoạn chương trình sau in ra giá trị nào?
def binary_search(a, x):
    left = 0
    right = len(a) - 1
    while left <= right:
        mid = (left + right) // 2
        if a[mid] == x:
            return mid
        if a[mid] < x:
            left = mid + 1
        else:
            right = mid - 1
    return -1

print(binary_search([1, 3, 5, 7, 9], 7))
A. 1    B. 2    C. 3    D. 4
